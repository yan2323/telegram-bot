# ---------- 資料庫部分 ----------
def init_db():
    conn = sqlite3.connect('shop.db')
    c = conn.cursor()
    c.execute('''CREATE TABLE IF NOT EXISTS sales
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  item TEXT,
                  price REAL,
                  box TEXT,
                  owner TEXT,
                  date TEXT)''')
    try:
        c.execute("ALTER TABLE sales ADD COLUMN owner TEXT DEFAULT 'Y'")
    except sqlite3.OperationalError:
        pass
    conn.commit()
    conn.close()

def add_sale(item, price, box, owner, date_str=None):
    if date_str:
        try:
            dt = datetime.strptime(date_str, "%Y-%m-%d")
            date_time_str = dt.strftime("%Y-%m-%d 00:00")
        except ValueError:
            try:
                dt = datetime.strptime(date_str, "%Y-%m-%d %H:%M")
                date_time_str = dt.strftime("%Y-%m-%d %H:%M")
            except ValueError:
                date_time_str = datetime.now().strftime("%Y-%m-%d %H:%M")
    else:
        date_time_str = datetime.now().strftime("%Y-%m-%d %H:%M")
    
    conn = sqlite3.connect('shop.db')
    c = conn.cursor()
    c.execute("INSERT INTO sales (item, price, box, owner, date) VALUES (?,?,?,?,?)",
              (item, price, box, owner, date_time_str))
    conn.commit()
    conn.close()

def get_adjusted_totals(cursor, where_clause="1=1"):
    cursor.execute(f"SELECT SUM(price) FROM sales WHERE owner='Y' AND {where_clause}")
    y_orig = cursor.fetchone()[0] or 0
    cursor.execute(f"SELECT SUM(price) FROM sales WHERE owner='C' AND {where_clause}")
    c_orig = cursor.fetchone()[0] or 0
    cursor.execute(f"SELECT SUM(price) FROM sales WHERE owner='A' AND {where_clause}")
    a_orig = cursor.fetchone()[0] or 0
    
    y_total = y_orig + c_orig + a_orig / 2
    n_total = (cursor.execute(f"SELECT SUM(price) FROM sales WHERE owner='N' AND {where_clause}").fetchone()[0] or 0) + a_orig / 2
    
    cursor.execute(f"SELECT COUNT(*) FROM sales WHERE owner='Y' AND {where_clause}")
    y_count = cursor.fetchone()[0] or 0
    cursor.execute(f"SELECT COUNT(*) FROM sales WHERE owner='C' AND {where_clause}")
    c_count = cursor.fetchone()[0] or 0
    cursor.execute(f"SELECT COUNT(*) FROM sales WHERE owner='A' AND {where_clause}")
    a_count = cursor.fetchone()[0] or 0
    cursor.execute(f"SELECT COUNT(*) FROM sales WHERE owner='N' AND {where_clause}")
    n_count = cursor.fetchone()[0] or 0
    
    return {
        'Y': {'total': y_total, 'count': y_count + c_count + a_count},
        'N': {'total': n_total, 'count': n_count + a_count}
    }

        "/batch                   - 批量匯入多筆舊帳\n"
        "/today                   - 查看今日營收 (自動分帳)\n"
        "/day YYYY-MM-DD          - 查看指定日期銷量\n"
        "/month [YYYY-MM]         - 查看月度報表\n"
        "/rank                    - 查看格子銷售排名\n"
        "/owner 歸屬              - 查原始歸屬總額 (Y/N/A/C)\n"
        "/actual Y/N              - 查實際分帳後老闆實收\n"
        "/export                  - 匯出全部原始資料為CSV"

async def add(update: Update, context: ContextTypes.DEFAULT_TYPE):
    try:
        args = context.args
        if len(args) < 4:
            await update.message.reply_text(
                "❌ 格式：/add 商品名 價格 格子號 歸屬(Y/N/A/C) [日期(YYYY-MM-DD)]\n"
                "例如：/add 鑰匙圈 50 A3 Y 2026-07-15"
            )
            return
        item = args[0]
        price = float(args[1])
        box = args[2]
        owner = args[3].upper()
        if owner not in ['Y', 'N', 'A', 'C']:
            await update.message.reply_text("❌ 歸屬只能是 Y(老闆Y), N(老闆N), A(集資), C(代C)")
            return
        date_str = None
        if len(args) >= 5:
            date_str = args[4]
        add_sale(item, price, box, owner, date_str)
        display_date = date_str if date_str else "今天"
        await update.message.reply_text(f"✅ 已記錄：{item} | ${price} | 格子{box} | 歸屬{owner} | 日期 {display_date}")
    except ValueError:
        await update.message.reply_text("❌ 價格必須是數字喔！")

async def batch(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text
    lines = text.split('\n')
    if lines and lines[0].strip().startswith('/batch'):
        lines = lines[1:]
    lines = [line.strip() for line in lines if line.strip()]
    if not lines:
        await update.message.reply_text(
            "❌ 請在 /batch 下方貼上多筆資料，每行一筆。\n\n"
            "格式：商品 價格 格子 歸屬 [日期]\n"
            "例如：\n鑰匙圈 50 A3 Y 2026-07-15\n公仔 120 B1 A 2026-08-03"
        )
        return
    success_count = 0
    fail_count = 0
    error_msgs = []
    for line in lines:
        parts = line.split()
        if len(parts) < 4:
            fail_count += 1
            error_msgs.append(f"❌ 欄位不足: {line}")
            continue
        try:
            item = parts[0]
            price = float(parts[1])
            box = parts[2]
            owner = parts[3].upper()
            if owner not in ['Y', 'N', 'A', 'C']:
                fail_count += 1
                error_msgs.append(f"❌ 歸屬錯誤: {line}")
                continue
            date_str = parts[4] if len(parts) >= 5 else None
            add_sale(item, price, box, owner, date_str)
            success_count += 1
        except ValueError:
            fail_count += 1
            error_msgs.append(f"❌ 價格非數字: {line}")
        except Exception as e:
            fail_count += 1
            error_msgs.append(f"❌ 其他錯誤: {line}")
    reply_msg = f"✅ 批量匯入完成！成功 {success_count} 筆，失敗 {fail_count} 筆。"
    if error_msgs and len(error_msgs) <= 5:
        reply_msg += "\n\n錯誤明細：\n" + "\n".join(error_msgs)
    elif error_msgs:
        reply_msg += f"\n\n(共 {len(error_msgs)} 筆錯誤，請檢查格式)"
    await update.message.reply_text(reply_msg)

async def today(update: Update, context: ContextTypes.DEFAULT_TYPE):
    conn = sqlite3.connect('shop.db')
    c = conn.cursor()
    today_str = datetime.now().strftime("%Y-%m-%d")
    where = f"date LIKE '{today_str}%'"
    
    c.execute(f"SELECT owner, COUNT(*), SUM(price) FROM sales WHERE {where} GROUP BY owner")
    rows = c.fetchall()
    total_count = sum(r[1] for r in rows)
    total_amt = sum(r[2] or 0 for r in rows)
    
    msg = f"📊 今日({today_str})總營收：共 {total_count} 單，總金額 ${total_amt:.2f}\n"
    msg += "━━━━━━━━━━━━\n"
    if rows:
        msg += "👥 原始分類業績：\n"
        for owner, cnt, amt in rows:
            amt = amt if amt else 0
            msg += f"  {owner} → {cnt}單 / ${amt:.2f}\n"
    else:
        msg += "暫無銷售記錄\n"
    
    adj = get_adjusted_totals(c, where)
    msg += "━━━━━━━━━━━━\n"
    msg += f"💰 實際分帳：\n"
    msg += f"  👤 老闆Y實收：${adj['Y']['total']:.2f} (含Y全額 + C全額 + A一半)\n"
    msg += f"  👤 老闆N實收：${adj['N']['total']:.2f} (含N全額 + A一半)\n"
    conn.close()
    await update.message.reply_text(msg)

async def day(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not context.args:
        await update.message.reply_text("❌ 請指定日期，格式：/day 2026-09-04")
        return
    target_date = context.args[0]
    if len(target_date) != 10 or target_date[4] != '-' or target_date[7] != '-':
        await update.message.reply_text("❌ 日期格式錯誤！請用 YYYY-MM-DD")
        return
    conn = sqlite3.connect('shop.db')
    c = conn.cursor()
    where = f"date LIKE '{target_date}%'"
    c.execute(f"SELECT COUNT(*), SUM(price) FROM sales WHERE {where}")
    count, total = c.fetchone()
    total = total if total else 0
    if not count or count == 0:
        await update.message.reply_text(f"📭 {target_date} 暫無銷售記錄")
        conn.close()
        return
    msg = f"📅 {target_date} 銷售報表\n"
    msg += "━━━━━━━━━━━━\n"
    msg += f"📈 總單數：{count} 筆\n💰 總金額：${total:.2f}\n"
    msg += "━━━━━━━━━━━━\n"
    c.execute(f"SELECT owner, COUNT(*), SUM(price) FROM sales WHERE {where} GROUP BY owner ORDER BY owner")
    rows = c.fetchall()
    if rows:
        msg += "👥 原始分類業績：\n"
        for owner, cnt, amt in rows:
            amt = amt if amt else 0
            msg += f"  {owner} → {cnt}單 / ${amt:.2f}\n"
    adj = get_adjusted_totals(c, where)
    msg += "━━━━━━━━━━━━\n"
    msg += f"💰 實際分帳：\n"
    msg += f"  👤 老闆Y實收：${adj['Y']['total']:.2f}\n"
    msg += f"  👤 老闆N實收：${adj['N']['total']:.2f}\n"
    msg += "━━━━━━━━━━━━\n"
    c.execute(f"SELECT item, price, box, owner, date FROM sales WHERE {where} ORDER BY date")
    details = c.fetchall()
    msg += "📋 明細清單：\n"
    for item, price, box, owner, dt in details:
        time_part = dt.split()[1] if ' ' in dt else dt
        msg += f"  {time_part} | {item} | ${price} | {box} | {owner}\n"
        if len(msg) > 3800:
            msg += "...(資料過多，請用 /export 匯出完整檔案查看)"
            break
    conn.close()
    await update.message.reply_text(msg)

async def rank(update: Update, context: ContextTypes.DEFAULT_TYPE):
    conn = sqlite3.connect('shop.db')
    c = conn.cursor()
    c.execute("SELECT box, SUM(price), COUNT(*) FROM sales GROUP BY box ORDER BY SUM(price) DESC")
    results = c.fetchall()
    conn.close()
    if not results:
        await update.message.reply_text("暫無數據")
        return
    msg = "🏆 格子銷售排名：\n"
    for i, (box, total, count) in enumerate(results, 1):
        msg += f"{i}. 格子{box} → 總${total:.2f} ({count}單)\n"
    await update.message.reply_text(msg)

async def owner(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not context.args:
        await update.message.reply_text("請指定原始歸屬，例如 /owner Y")
        return
    owner_code = context.args[0].upper()
    if owner_code not in ['Y', 'N', 'A', 'C']:
        await update.message.reply_text("歸屬只能是 Y, N, A, C")
        return
    conn = sqlite3.connect('shop.db')
    c = conn.cursor()
    c.execute("SELECT COUNT(*), SUM(price) FROM sales WHERE owner=?", (owner_code,))
    count, total = c.fetchone()
    conn.close()
    total = total if total else 0
    await update.message.reply_text(f"🧾 原始歸屬 {owner_code} 的總業績：共 {count} 單，總金額 ${total:.2f}")

async def actual(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not context.args:
        await update.message.reply_text("請指定老闆，例如 /actual Y")
        return
    boss = context.args[0].upper()
    if boss not in ['Y', 'N']:
        await update.message.reply_text("請輸入 Y 或 N")
        return
    conn = sqlite3.connect('shop.db')
    c = conn.cursor()
    adj = get_adjusted_totals(c)
    conn.close()
    await update.message.reply_text(f"👤 老闆{boss} 實收總額：${adj[boss]['total']:.2f} (包含分帳調整)")

async def export(update: Update, context: ContextTypes.DEFAULT_TYPE):
    conn = sqlite3.connect('shop.db')
    c = conn.cursor()
    c.execute("SELECT date, item, price, box, owner FROM sales ORDER BY date")
    rows = c.fetchall()
    conn.close()
    if not rows:
        await update.message.reply_text("📭 資料庫空空如也，沒有資料可以匯出")
        return
    output = io.BytesIO()
    output.write('\uFEFF'.encode('utf-8'))
    text_output = io.StringIO()
    writer = csv.writer(text_output)
    writer.writerow(['日期時間', '商品名', '價格', '格子號', '歸屬'])
    writer.writerows(rows)
    output.write(text_output.getvalue().encode('utf-8'))
    output.seek(0)
    await update.message.reply_document(
        document=output,
        filename="格子舖銷售報表.csv",
        caption="📊 全部原始銷售資料已匯出，請查收！"
    )

    bot_app = Application.builder().token(TOKEN).build()
    
    bot_app.add_handler(CommandHandler("start", start))
    bot_app.add_handler(CommandHandler("add", add))
    bot_app.add_handler(CommandHandler("batch", batch))
    bot_app.add_handler(CommandHandler("today", today))
    bot_app.add_handler(CommandHandler("day", day))
    bot_app.add_handler(CommandHandler("month", month))
    bot_app.add_handler(CommandHandler("rank", rank))
    bot_app.add_handler(CommandHandler("owner", owner))
    bot_app.add_handler(CommandHandler("actual", actual))
    bot_app.add_handler(CommandHandler("export", export))
    
    print("✅ 格子舖Bot已啟動！")
    bot_app.run_polling()
